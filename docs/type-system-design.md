## AI SDK Types Overview

### UIMessage

**Source**: `@ai-sdk/core` and `@ai-sdk/ui`

**Purpose**: Standardized message type for UI components

**Definition**:
```typescript
interface UIMessage<Metadata = Record<string, unknown>> {
  id: string;
  role: 'user' | 'assistant' | 'system';
  parts: Array<
    | UIMessageTextPart
    | UIMessageToolCallPart
    | UIMessageToolResultPart
  >;
  metadata?: Metadata;
}

// Text part
interface UIMessageTextPart {
  type: 'text';
  text: string;
}

// Tool call part
interface UIMessageToolCallPart {
  type: 'tool-call';
  toolCallId: string;
  toolName: string;
  args: Record<string, unknown>;
}

// Tool result part
interface UIMessageToolResultPart {
  type: 'tool-result';
  toolCallId: string;
  result: unknown;
}
```

**Key Characteristics**:
- Immutable by convention (but not enforced)
- Supports rich content (text, tool calls, tool results)
- Optional metadata for extensions
- Used throughout AI SDK UI ecosystem

### ModelMessage

**Source**: `ai` (from AI SDK Core)

**Purpose**: Fundamental message structure used with AI SDK Core functions (`generateText`, `streamText`, etc.).

**Definition**:
```typescript
type ModelMessage =
  | SystemModelMessage
  | UserModelMessage
  | AssistantModelMessage
  | ToolModelMessage;

/** System message: can contain system information */
interface SystemModelMessage {
  role: 'system';
  content: string;
}

/** User message: can contain text or a combination of text, images, and files */
interface UserModelMessage {
  role: 'user';
  content: string | Array<TextPart | ImagePart | FilePart>;
}

/** Assistant message: can contain text, tool calls, or a combination of both */
interface AssistantModelMessage {
  role: 'assistant';
  content: string | Array<TextPart | ToolCallPart>;
}

/** Tool message: contains the result of one or more tool calls */
interface ToolModelMessage {
  role: 'tool';
  content: Array<ToolResultPart>;
}
```

**Content Parts**:

```typescript
// Text content part
interface TextPart {
  type: 'text';
  text: string;
}

// Image content part
interface ImagePart {
  type: 'image';
  image: DataContent | URL; // base64 string, Uint8Array, Buffer, or URL
  mediaType?: string;      // recommended to leave out for auto-detection
}

// File content part
interface FilePart {
  type: 'file';
  data: DataContent | URL;
  filename?: string;
  mediaType: string;
}

// Tool call part (generated by AI)
interface ToolCallPart {
  type: 'tool-call';
  toolCallId: string;
  toolName: string;
  args: unknown; // JSON-serializable object matching input schema
}

// Tool result part (sent back to AI)
interface ToolResultPart {
  type: 'tool-result';
  toolCallId: string;
  toolName: string;
  output: ToolResultOutput;
}

// Tool Result Output variants
type ToolResultOutput = 
  | { type: 'text'; value: string }
  | { type: 'json'; value: JSONValue }
  | { type: 'error-text'; value: string }
  | { type: 'error-json'; value: JSONValue }
  | { type: 'content'; value: Array<TextPart | FileDataPart | ImageDataPart | ...> }
  | { type: 'execution-denied'; reason?: string };
```

**Key Characteristics**:
- Used internally by AI SDK Core
- High-fidelity representation for multi-modal and tool-calling scenarios
- Should be converted from `vscode.LanguageModelChatRequestMessage` or `UIMessage`
- **Important**: For durability and resilience against prompt injection, use the `system` property in `generateText` instead of a separate system message when possible.

### Tool

**Source**: `@ai-sdk/core`

**Purpose**: Define tools for AI SDK to call

**Definition**:
```typescript
interface Tool<RESULT = unknown> {
  description: string;
  inputSchema: z.ZodTypeAny;  // Zod schema for input validation
  execute: (args: z.infer<z.ZodTypeAny>) => RESULT | Promise<RESULT>;
}
```

**Usage**:
```typescript
const weatherTool = tool({
  description: 'Get the current weather in a location',
  inputSchema: z.object({
    location: z.string().describe('The city and state, e.g. San Francisco, CA'),
  }),
  execute: async ({ location }) => {
    return {
      location,
      temperature: 72,
      conditions: 'Sunny',
    };
  },
});
```

## VS Code API Compatibility

### Challenge

AI SDK uses `UIMessage`, but VS Code's `LanguageModelChatMessage` has a different format. We need to ensure Playground doesn't depend on VS Code API while maintaining compatibility for Copilot integration.

### Solution: Separate Conversion Layers

#### VS Code API Types (for Copilot Integration Only)

```typescript
import * as vscode from 'vscode';

// VS Code's LanguageModelChatMessage
// This is ONLY used for Copilot integration, not for Playground
type VSCodeLanguageModelMessage = vscode.LanguageModelChatMessage;
```

#### Conversion Functions (Copilot Integration Only)

```typescript
// src/core/llm/messageConverter.ts
import * as vscode from 'vscode';
import type { ModelMessage } from 'ai';

/**
 * Convert VS Code LanguageModelChatMessage[] to AI SDK ModelMessage[]
 * This is ONLY used for Copilot integration, not for Playground
 */
export function convertToModelMessages(
  messages: vscode.LanguageModelChatMessage[]
): ModelMessage[] {
  return messages.map(msg => {
    if (msg.role === vscode.LanguageModelChatMessageRole.User) {
      return {
        role: 'user',
        content: msg.content.map(part => {
          if (part instanceof vscode.LanguageModelTextPart) {
            return { type: 'text', text: part.value };
          }
          throw new Error('Unsupported part type');
        }),
      };
    } else if (msg.role === vscode.LanguageModelChatMessageRole.Assistant) {
      return {
        role: 'assistant',
        content: msg.content.map(part => {
          if (part instanceof vscode.LanguageModelTextPart) {
            return { type: 'text', text: part.value };
          }
          throw new Error('Unsupported part type');
        }),
      };
    } else if (msg.role === vscode.LanguageModelChatMessageRole.System) {
      return {
        role: 'system',
        content: msg.content as string,
      };
    }
    throw new Error('Unknown message role');
  });
}

/**
 * Convert AI SDK ModelMessage[] to VS Code LanguageModelChatMessage[]
 * This is ONLY used for Copilot integration, not for Playground
 */
export function convertFromModelMessages(
  messages: ModelMessage[]
): vscode.LanguageModelChatMessage[] {
  return messages.map(msg => {
    if (msg.role === 'user') {
      return vscode.LanguageModelChatMessage.User(
        msg.content.map(part => {
          if (part.type === 'text') {
            return vscode.LanguageModelTextPart.create(part.text);
          }
          throw new Error('Unsupported part type');
        })
      );
    } else if (msg.role === 'assistant') {
      return vscode.LanguageModelChatMessage.Assistant(
        msg.content.map(part => {
          if (part.type === 'text') {
            return vscode.LanguageModelTextPart.create(part.text);
          }
          throw new Error('Unsupported part type');
        })
      );
    } else if (msg.role === 'system') {
      return vscode.LanguageModelChatMessage.System(msg.content);
    }
    throw new Error('Unknown message role');
  });
}
```

**Important Notes**:
- These conversion functions are ONLY for Copilot integration
- Playground uses AI SDK's `convertToModelMessages()` directly
- No VS Code API dependency in Playground code

### Type Guards

**Purpose**: Runtime type checking for message parts

```typescript
import type { UIMessage } from 'ai';

function isTextPart(part: UIMessage['parts'][number]): part is { type: 'text'; text: string } {
  return part.type === 'text';
}

function isToolCallPart(part: UIMessage['parts'][number]): part is { type: 'tool-call'; toolCallId: string; toolName: string; args: any } {
  return part.type === 'tool-call';
}

function isToolResultPart(part: UIMessage['parts'][number]): part is { type: 'tool-result'; toolCallId: string; result: any } {
  return part.type === 'tool-result';
}

// Usage
messages.forEach(message => {
  message.parts.forEach(part => {
    if (isTextPart(part)) {
      console.log('Text:', part.text);
    } else if (isToolCallPart(part)) {
      console.log('Tool called:', part.toolName, part.args);
    } else if (isToolResultPart(part)) {
      console.log('Tool result:', part.result);
    }
  });
});
```

## Storage and Persistence Types

### JSON Serialization

**Challenge**: UIMessage parts may contain non-serializable types (e.g., Zod schemas)

**Solution**: Strip non-serializable parts before storage

```typescript
function sanitizeForStorage(message: PlaygroundUIMessage): StoredMessage {
  return {
    id: message.id,
    role: message.role,
    parts: message.parts.map(part => {
      if (part.type === 'text') {
        return part;
      } else if (part.type === 'tool-call') {
        return {
          type: 'tool-call',
          toolCallId: part.toolCallId,
          toolName: part.toolName,
          args: JSON.parse(JSON.stringify(part.args)), // Deep clone
        };
      } else if (part.type === 'tool-result') {
        return {
          type: 'tool-result',
          toolCallId: part.toolCallId,
          result: JSON.parse(JSON.stringify(part.result)), // Deep clone
        };
      }
      throw new Error('Unknown part type');
    }),
    metadata: message.metadata,
    createdAt: message.metadata?.createdAt ?? Date.now(),
  };
}
```

### Compression

**Purpose**: Reduce storage size for long conversations

```typescript
import { compressToUTF16, decompressFromUTF16 } from 'lz-string';

function compressMessage(message: StoredMessage): string {
  return compressToUTF16(JSON.stringify(message));
}

function decompressMessage(compressed: string): StoredMessage {
  return JSON.parse(decompressFromUTF16(compressed));
}
```

### Encryption

**Purpose**: Encrypt sensitive messages before storage

```typescript
import { createCipheriv, createDecipheriv } from 'crypto';

function encryptMessage(message: StoredMessage, key: Buffer, iv: Buffer): string {
  const cipher = createCipheriv('aes-256-cbc', key, iv);
  let encrypted = cipher.update(JSON.stringify(message), 'utf8', 'hex');
  encrypted += cipher.final('hex');
  return encrypted;
}

function decryptMessage(encrypted: string, key: Buffer, iv: Buffer): StoredMessage {
  const decipher = createDecipheriv('aes-256-cbc', key, iv);
  let decrypted = decipher.update(encrypted, 'hex', 'utf8');
  decrypted += decipher.final('utf8');
  return JSON.parse(decrypted);
}
```

## Migration Guide

### Step 1: Replace Custom ChatMessage Type

**Before**:
```typescript
// Old custom type
export interface ChatMessage {
  id: string;
  role: 'user' | 'assistant' | 'system';
  content: string;
  timestamp?: number;
  model?: string;
}
```

**After**:
```typescript
// New AI SDK type
import { UIMessage } from 'ai';

export interface PlaygroundMetadata {
  createdAt: number;
  modelSid: string;
  temperature: number;
  // ... other metadata
}

export type PlaygroundUIMessage = UIMessage<PlaygroundMetadata>;
```

**Action**:
1. Add AI SDK dependency: `npm install ai`
2. Update type definitions
3. Update all usages from `ChatMessage` to `PlaygroundUIMessage`

### Step 2: Update Content Access

**Before**:
```typescript
const message: ChatMessage = { ... };
console.log(message.content); // Direct string access
```

**After**:
```typescript
const message: PlaygroundUIMessage = { ... };
// Iterate through parts
message.parts.forEach(part => {
  if (part.type === 'text') {
    console.log(part.text);
  }
});

// Or get first text part
const textContent = message.parts
  .filter((part): part is { type: 'text'; text: string } => part.type === 'text')
  .map(part => part.text)
  .join('\n');
```

**Helper Function**:
```typescript
function getMessageText(message: PlaygroundUIMessage): string {
  return message.parts
    .filter((part): part is { type: 'text'; text: string } => part.type === 'text')
    .map(part => part.text)
    .join('\n');
}

// Usage
const text = getMessageText(message);
console.log(text);
```

### Step 5: Update Streaming Logic

**Before**:
```typescript
const response = await chatModel.sendRequest(messages, {});
const stream = response.stream;

let assistantContent = '';
for await (const chunk of stream) {
  if (chunk.type === 'text') {
    assistantContent += chunk.value;
  }
}

const assistantMessage: ChatMessage = {
  id: generateId(),
  role: 'assistant',
  content: assistantContent,
  timestamp: Date.now(),
  model: 'gpt-4',
};
```

**After**:
```typescript
import { readUIMessageStream, generateId } from 'ai';

let currentAssistantMessage: PlaygroundUIMessage | null = null;

for await (const uiMessage of readUIMessageStream({
  stream: result.toUIMessageStream(),
  originalMessages: history,
})) {
  if (uiMessage.role === 'assistant') {
    if (!currentAssistantMessage) {
      // First chunk
      currentAssistantMessage = {
        id: generateId(),
        role: 'assistant',
        parts: uiMessage.parts,
        metadata: {
          createdAt: Date.now(),
          modelSid: 'gpt-4',
          temperature: 0.7,
          isStreaming: true,
        },
      };
    } else {
      // Update streaming content
      currentAssistantMessage.parts = uiMessage.parts;
    }
  }
}

// Mark as completed
if (currentAssistantMessage) {
  currentAssistantMessage.metadata.isStreaming = false;
}
```

```typescript
// Or send directly (UIMessage is JSON-serializable except for special cases)
webview.postMessage({
  type: 'message',
  message: {
    id: message.id,
    role: message.role,
    parts: message.parts,
    metadata: message.metadata,
  },
});
```
